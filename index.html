Security Lab & Code Comparison

Interactive demonstration of vulnerable code vs. secure PDO prepared statements.

auth_handler.php
// ❌ HIGH RISK: Unsanitized Direct String Concatenation (Vulnerable to SQL Injection)
$username = $_POST['username'];
$password = $_POST['password'];
// Vulnerable SQL Query concatenation:
$query = "SELECT * FROM users WHERE user = '" . $username . "' AND pass = '" . $password . "'";
$result = mysqli_query($conn, $query);
// ✅ HARDENED: PDO Prepared Statement with Parameter Binding
$stmt = $pdo->prepare('SELECT id, password_hash FROM users WHERE username = :user LIMIT 1');
$stmt->execute([':user' => $username]);
$user = $stmt->fetch();
// Verify password hash securely:
if ($user && password_verify($password, $user['password_hash'])) {
    session_regenerate_id(true);
}
